            generateQRCode(qrString) {
                const container = document.getElementById('qrCodeContainer');
                container.innerHTML = ''; // Clear previous QR
                
                if (!qrString) {
                    container.innerHTML = '<p class="text-red-500">QR code tidak tersedia</p>';
                    return;
                }

                try {
                    // Gunakan qrcode-generator library
                    const typeNumber = 0; // Auto
                    const errorCorrectionLevel = 'L';
                    const qr = qrcode(typeNumber, errorCorrectionLevel);
                    qr.addData(qrString);
                    qr.make();
                    
                    // Create canvas element
                    const canvas = document.createElement('canvas');
                    const size = 256;
                    const cellSize = size / qr.getModuleCount();
                    const ctx = canvas.getContext('2d');
                    
                    canvas.width = size;
                    canvas.height = size;
                    
                    // Draw QR code
                    for (let row = 0; row < qr.getModuleCount(); row++) {
                        for (let col = 0; col < qr.getModuleCount(); col++) {
                            ctx.fillStyle = qr.isDark(row, col) ? '#000000' : '#FFFFFF';
                            ctx.fillRect(
                                col * cellSize,
                                row * cellSize,
                                cellSize,
                                cellSize
                            );
                        }
                    }
                    
                    container.appendChild(canvas);
                    
                } catch (error) {
                    console.error('Error generating QR code:', error);
                    // Fallback: tampilkan QR string sebagai text (untuk debugging)
                    container.innerHTML = `
                        <div class="text-center">
                            <p class="text-red-500 mb-2">Error generating QR code</p>
                            <p class="text-xs text-gray-500 break-all">${qrString.substring(0, 100)}...</p>
                        </div>
                    `;
                }
            }

            updateServerInfo(result) {
                document.getElementById('serverStatus').textContent = result.success ? 'Connected' : 'Disconnected';
                document.getElementById('lastUpdate').textContent = new Date().toLocaleTimeString('id-ID');

                // Tambahkan warna berdasarkan status
                const serverStatus = document.getElementById('serverStatus');
                serverStatus.className = 'font-medium ' + (result.success ? 'text-green-600' : 'text-red-600');
            }

            async syncStatus() {

                        method: 'POST',

                            'X-CSRF-TOKEN': '{{ csrf_token() }}'
                        }
                    });

                    const result = await response.json();
                    
                    if (result.success) {
                        this.showMessage('Status berhasil di-sync', 'success');
                        this.loadStatus(true);
                    } else {
                        this.showMessage('Gagal sync status: ' + result.message, 'error');
                    }
                } catch (error) {
                    console.error('Error syncing status:', error);
                    this.showMessage('Terjadi kesalahan saat sync status', 'error');
                }
            }

            async restartBot() {
                if (!confirm('Apakah Anda yakin ingin merestart WhatsApp bot?')) {
                    return;
                }

                try {
                    this.showMessage('Merestart WhatsApp bot...', 'info');
                    
                    const response = await fetch(this.restartEndpoint, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                            'X-CSRF-TOKEN': '{{ csrf_token() }}'
                        }
                    });

                    const result = await response.json();
                    
                    if (result.success) {
                        this.showMessage('WhatsApp bot restart berhasil', 'success');
                        setTimeout(() => this.loadStatus(true), 2000);
                    } else {
                        this.showMessage('Gagal restart bot: ' + result.message, 'error');
                    }
                } catch (error) {
                    console.error('Error restarting bot:', error);
                    this.showMessage('Terjadi kesalahan saat restart bot', 'error');
                }
            }

            downloadQR() {
                const canvas = document.querySelector('#qrCodeContainer canvas');
                if (canvas) {
                    const link = document.createElement('a');
                    link.download = 'whatsapp-qr-code.png';
                    link.href = canvas.toDataURL('image/png');
                    link.click();
                } else {
                    this.showMessage('Tidak ada QR code untuk di-download', 'error');
                }
            }

            testConnection() {
                this.showMessage('Testing koneksi server...', 'info');
                this.loadStatus(true);
            }

            startPolling() {
                // Poll setiap 10 detik untuk status terbaru
                this.pollingInterval = setInterval(() => {
                    this.loadStatus();
                }, 10000);
            }

            stopPolling() {
                if (this.pollingInterval) {
                    clearInterval(this.pollingInterval);
                }
            }

            showMessage(message, type = 'info') {
                const toast = document.getElementById('toast');
                const toastIcon = document.getElementById('toastIcon');
                const toastMessage = document.getElementById('toastMessage');
                
                // Set warna dan icon berdasarkan type
                const styles = {
                    success: { bg: 'bg-green-500', icon: 'fas fa-check-circle' },
                    error: { bg: 'bg-red-500', icon: 'fas fa-exclamation-circle' },
                    info: { bg: 'bg-blue-500', icon: 'fas fa-info-circle' },
                    warning: { bg: 'bg-yellow-500', icon: 'fas fa-exclamation-triangle' }
                };
                
                const style = styles[type] || styles.info;
                
                toast.className = `fixed top-4 right-4 p-4 rounded-lg shadow-lg text-white max-w-sm ${style.bg}`;
                toastIcon.className = style.icon;
                toastMessage.textContent = message;
                
                toast.classList.remove('hidden');
                
                // Auto hide setelah 5 detik
                setTimeout(() => {
                    this.hideToast();
                }, 5000);
            }

            hideToast() {
                const toast = document.getElementById('toast');
                toast.classList.add('hidden');
            }

            showError(message) {
                this.showMessage(message, 'error');
            }
        }

        // Global function untuk hide toast
        function hideToast() {
            const toast = document.getElementById('toast');
            toast.classList.add('hidden');
