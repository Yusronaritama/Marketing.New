public function getUsersFromMasterMoodle(Request $request)
    {
        try {
            $limit = $request->input('limit', 50);
            $status = $request->input('status', 'success');

            $result = $this->moodleModel->getUsersFromMasterMoodle($limit, $status);

            // SELALU return JSON untuk AJAX request
            if ($request->ajax() || $request->wantsJson() || $request->is('api/*')) {
                // Jika perlu data tambahan untuk AJAX, tambahkan di sini
                if ($result['success'] && !empty($result['users'])) {
                    $usersWithKelulusan = [];
                    foreach ($result['users'] as $user) {
                        $kelulusanData = $this->getKelulusanData($user['master_user_id']);
                        $usersWithKelulusan[] = array_merge($user, $kelulusanData);
                    }
                    $result['users'] = $usersWithKelulusan;
                }
                return response()->json($result);
            }

            // Untuk view HTML biasa
            if ($result['success']) {
                // Ambil data kelulusan untuk setiap user
                $usersWithKelulusan = [];
                foreach ($result['users'] as $user) {
                    $kelulusanData = $this->getKelulusanData($user['master_user_id']);
                    $usersWithKelulusan[] = array_merge($user, $kelulusanData);
                }
                
                $result['users'] = $usersWithKelulusan;

                return view('admin.moodle.master-moodle-users', [
                    'users' => $result['users'],
                    'total_users' => $result['total_users'],
                    'limit' => $limit,
                    'status' => $status,
                    'success' => true
                ]);
            } else {
                return view('admin.moodle.master-moodle-users', [
                    'error' => $result['error'],
                    'message' => $result['message'] ?? 'Gagal mengambil data',
                    'limit' => $limit,
                    'status' => $status,
                    'success' => false
                ]);
            }

        } catch (\Exception $e) {
            Log::error('Error getting users from master_moodle: ' . $e->getMessage());
            
            // Untuk AJAX request
            if ($request->ajax() || $request->wantsJson() || $request->is('api/*')) {
                return response()->json([
                    'success' => false,
                    'error' => $e->getMessage()
                ]);
            }
            
            return view('admin.moodle.master-moodle-users', [
                'error' => 'Terjadi kesalahan sistem: ' . $e->getMessage(),
                'success' => false
            ]);
        }
    }
 public function syncToMasterMoodleUsers(Request $request)
{
    try {
        DB::beginTransaction();

        $syncType = $request->input('sync_type', 'all');
        $usersData = $request->input('users_data', []);
        
        // Jika tidak ada data yang dikirim, ambil dari fungsi getUsersFromMasterMoodle
        if (empty($usersData)) {
            $result = $this->getUsersFromMasterMoodle(new Request(['limit' => 1000]));
            
            if (!$result['success']) {
                throw new \Exception('Gagal mengambil data users: ' . ($result['error'] ?? 'Unknown error'));
            }
            
            $usersData = $result['users'];
        }

        $stats = [
            'total' => count($usersData),
            'created' => 0,
            'updated' => 0,
            'skipped' => 0,
            'errors' => 0
        ];

        foreach ($usersData as $user) {
            try {
                // Pastikan data yang diperlukan tersedia
                if (empty($user['master_user_id']) || empty($user['moodle_user_id'])) {
                    $stats['skipped']++;
                    Log::warning('Skipping user - missing master_user_id or moodle_user_id', $user);
                    continue;
                }

                // Cek apakah user sudah ada di master_moodle_users
                $existingUser = DB::table('master_moodle_users')
                                ->where('master_user_id', $user['master_user_id'])
                                ->orWhere('moodle_user_id', $user['moodle_user_id'])
                                ->first();

                // Siapkan data untuk insert/update
                $dataToInsert = [
                    'master_user_id' => $user['master_user_id'],
                    'moodle_user_id' => $user['moodle_user_id'],
                    'username' => $user['username'] ?? '',
                    'email' => $user['email'] ?? '',
                    'firstname' => $user['nama'] ?? '', // Gunakan 'nama' dari data
                    'lastname' => '', // Default kosong
                    'updated_at' => now()
                ];

                // Tambahkan data quiz berdasarkan sync_type - PERBAIKAN DI SINI
                if ($syncType === 'all' || $syncType === 'quiz_only') {
                    $dataToInsert['quiz_id'] = $user['quiz_id'] ?? 1197; // Default quiz ID
                    
                    // PRIORITAS: Ambil nilai dari berbagai sumber yang mungkin
                    $nilai = $user['nilai'] ?? $user['nilai_quiz'] ?? 0;
                    $dataToInsert['nilai_quiz'] = $nilai;
                    
                    $dataToInsert['attempt_count'] = $user['attempt_count'] ?? 0;
                    
                    Log::info('Setting quiz data for user', [
                        'master_user_id' => $user['master_user_id'],
                        'nilai_sources' => [
                            'user[nilai]' => $user['nilai'] ?? 'NOT SET',
                            'user[nilai_quiz]' => $user['nilai_quiz'] ?? 'NOT SET',
                            'final_nilai' => $nilai
                        ],
                        'attempt_count' => $user['attempt_count'] ?? 0
                    ]);
                }

                // Tambahkan data kelulusan berdasarkan sync_type
                if ($syncType === 'all' || $syncType === 'kelulusan_only') {
                    // Gunakan data dari getKelulusanData yang sudah diproses
                    $dataToInsert['status_kelulusan'] = $user['status_kelulusan'] ?? 'Belum diketahui';
                    $dataToInsert['prodi_lulus'] = $user['prodi_lulus'] ?? '';
                    $dataToInsert['nama_prodi_lulus'] = $user['nama_prodi_lulus'] ?? '';
                    $dataToInsert['pilihan_lulus'] = $user['pilihan_lulus'] ?? '';
                    
                    // Hitung selisih nilai jika ada data kelulusan info
                    if (isset($user['kelulusan_info']) && isset($user['kelulusan_info']['selisih'])) {
                        $dataToInsert['selisih_nilai'] = $user['kelulusan_info']['selisih'];
                    } else {
                        $dataToInsert['selisih_nilai'] = $user['selisih_nilai'] ?? null;
                    }
                }

                // Tambahkan data prodi pilihan
                $dataToInsert['prodi_pilihan1'] = $user['prodi_pilihan1'] ?? '';
                $dataToInsert['nama_prodi_pilihan1'] = $user['nama_prodi_pilihan1'] ?? '';
                $dataToInsert['prodi_pilihan2'] = $user['prodi_pilihan2'] ?? '';
                $dataToInsert['nama_prodi_pilihan2'] = $user['nama_prodi_pilihan2'] ?? '';
                $dataToInsert['data_source'] = $user['data_source'] ?? 'manual';

                if ($existingUser) {
                    // Update existing record
                    DB::table('master_moodle_users')
                        ->where('id', $existingUser->id)
                        ->update($dataToInsert);
                    $stats['updated']++;
                    
                    Log::info('Updated user in master_moodle_users', [
                        'master_user_id' => $user['master_user_id'],
                        'moodle_user_id' => $user['moodle_user_id'],
                        'nilai_quiz' => $dataToInsert['nilai_quiz'] ?? 0
                    ]);
                } else {
                    // Insert new record
                    $dataToInsert['created_at'] = now();
                    DB::table('master_moodle_users')->insert($dataToInsert);
                    $stats['created']++;
                    
                    Log::info('Created user in master_moodle_users', [
                        'master_user_id' => $user['master_user_id'],
                        'moodle_user_id' => $user['moodle_user_id'],
                        'nilai_quiz' => $dataToInsert['nilai_quiz'] ?? 0
                    ]);
                }

            } catch (\Exception $e) {
                Log::error('Error syncing user to master_moodle_users: ' . $e->getMessage(), [
                    'user_data' => $user
                ]);
                $stats['errors']++;
            }
        }

        DB::commit();

        Log::info('Sync to master_moodle_users completed', $stats);

        return response()->json([
            'success' => true,
            'message' => 'Data berhasil disinkronisasi ke master_moodle_users',
            'stats' => $stats,
            'sync_type' => $syncType
        ]);

    } catch (\Exception $e) {
        DB::rollBack();
        
        Log::error('Error syncing to master_moodle_users: ' . $e->getMessage());

        return response()->json([
            'success' => false,
            'error' => 'Gagal menyinkronisasi data: ' . $e->getMessage()
        ]);
    }
}

 public function getMasterMoodleUsers(Request $request)
    {
        try {
            $limit = $request->input('limit', 50);
            $search = $request->input('search', '');
            $status = $request->input('status', '');

            $query = DB::table('master_moodle_users')
                     ->select('*')
                     ->orderBy('updated_at', 'desc');

            if (!empty($search)) {
                $query->where(function($q) use ($search) {
                    $q->where('username', 'like', "%{$search}%")
                      ->orWhere('email', 'like', "%{$search}%")
                      ->orWhere('firstname', 'like', "%{$search}%");
                });
            }

            if (!empty($status)) {
                $query->where('status_kelulusan', $status);
            }

            $users = $query->paginate($limit);

            $summary = [
                'total' => DB::table('master_moodle_users')->count(),
                'lulus' => DB::table('master_moodle_users')->where('status_kelulusan', 'like', '%LULUS%')->count(),
                'tidak_lulus' => DB::table('master_moodle_users')->where('status_kelulusan', 'like', '%TIDAK LULUS%')->count(),
                'belum_diketahui' => DB::table('master_moodle_users')->where('status_kelulusan', 'Belum diketahui')->count(),
                'punya_nilai' => DB::table('master_moodle_users')->where('nilai_quiz', '>', 0)->count()
            ];

            if ($request->ajax() || $request->wantsJson()) {
                return response()->json([
                    'success' => true,
                    'users' => $users,
                    'summary' => $summary,
                    'pagination' => [
                        'total' => $users->total(),
                        'per_page' => $users->perPage(),
                        'current_page' => $users->currentPage(),
                        'last_page' => $users->lastPage()
                    ]
                ]);
            }

            return view('admin.moodle.master-moodle-users-list', [
                'users' => $users,
                'summary' => $summary,
                'search' => $search,
                'status' => $status,
                'limit' => $limit,
                'success' => true
            ]);

        } catch (\Exception $e) {
            Log::error('Error getting master_moodle_users: ' . $e->getMessage());

            if ($request->ajax() || $request->wantsJson()) {
                return response()->json([
                    'success' => false,
                    'error' => $e->getMessage()
                ]);
            }

            return view('admin.moodle.master-moodle-users-list', [
                'error' => 'Terjadi kesalahan: ' . $e->getMessage(),
                'success' => false
            ]);
        }
    }

private function getNilaiQuizTerbaru($userId)
{
    try {
        // Default quiz ID
        $quizId = 1197;
        
        // Pastikan moodleRest tersedia
        $moodleRest = $this->moodleModel->getMoodleRest();
        if (!$moodleRest) {
            Log::error('MoodleRest not available for user_id: ' . $userId);
            return null;
        }

        // Ambil attempts dari Moodle
        $attemptParams = [
            'quizid' => (int)$quizId,
            'userid' => (int)$userId
        ];

        Log::info('Getting quiz attempts for user', [
            'user_id' => $userId,
            'quiz_id' => $quizId
        ]);

        $attempts = $moodleRest->request('mod_quiz_get_user_attempts', $attemptParams);

        if (isset($attempts['exception'])) {
            Log::warning('Error getting quiz attempts for user: ' . $userId . ' - ' . $attempts['exception']);
            return null;
        }

        if (!isset($attempts['attempts']) || empty($attempts['attempts'])) {
            Log::info('No quiz attempts found for user: ' . $userId);
            return null;
        }

        // Cari attempt terbaru yang sudah selesai
        $latestAttempt = null;
        $highestGrade = 0;

        foreach ($attempts['attempts'] as $attempt) {
            if (($attempt['state'] === 'finished') && isset($attempt['sumgrades'])) {
                if ($attempt['sumgrades'] > $highestGrade) {
                    $highestGrade = $attempt['sumgrades'];
                    $latestAttempt = $attempt;
                }
            }
        }

        // Jika tidak ada yang finished, ambil attempt terakhir
        if (!$latestAttempt && !empty($attempts['attempts'])) {
            $latestAttempt = end($attempts['attempts']);
            $highestGrade = $latestAttempt['sumgrades'] ?? 0;
        }

        if ($latestAttempt) {
            $result = [
                'nilai' => $highestGrade,
                'quiz_id' => $quizId,
                'attempt_id' => $latestAttempt['id'],
                'state' => $latestAttempt['state'] ?? 'unknown'
            ];

            // Tambahkan tanggal attempt jika tersedia
            if (isset($latestAttempt['timestart'])) {
                $result['attempt_date'] = \Carbon\Carbon::createFromTimestamp($latestAttempt['timestart'])->format('Y-m-d H:i:s');
            }

            Log::info('Quiz result found for user', [
                'user_id' => $userId,
                'nilai' => $highestGrade,
                'attempt_id' => $latestAttempt['id']
            ]);

            return $result;
        }

        Log::info('No valid quiz attempts found for user: ' . $userId);
        return null;

    } catch (\Exception $e) {
        Log::error('Error getting nilai quiz for user_id ' . $userId . ': ' . $e->getMessage());
        return null;
    }
}

private function tentukanPilihanLulus($userId, $prodiLulus)
{
    try {
        // Ambil data prodi pilihan dari master_ereg
        $masterEreg = DB::table('master_ereg')
                     ->where('user_id', $userId)
                     ->select('jurusan1', 'jurusan2')
                     ->first();

        if (!$masterEreg) {
            return 'Tidak diketahui';
        }

        // Bandingkan prodi lulus dengan pilihan 1 dan 2
        if ($masterEreg->jurusan1 == $prodiLulus) {
            return 'Pilihan 1';
        } elseif ($masterEreg->jurusan2 == $prodiLulus) {
            return 'Pilihan 2';
        } else {
            // Jika tidak cocok dengan pilihan 1 atau 2, coba mapping
            $prodiMapping = $this->getProdiMapping();
            $kodePilihan1 = $prodiMapping[$masterEreg->jurusan1] ?? $masterEreg->jurusan1;
            $kodePilihan2 = $prodiMapping[$masterEreg->jurusan2] ?? $masterEreg->jurusan2;
            
            if ($kodePilihan1 == $prodiLulus) {
                return 'Pilihan 1';
            } elseif ($kodePilihan2 == $prodiLulus) {
                return 'Pilihan 2';
            } else {
                return 'Prodi Lain';
            }
        }

    } catch (\Exception $e) {
        Log::error('Error menentukan pilihan lulus: ' . $e->getMessage());
        return 'Error';
    }
}
private function getPassingGrade($kodeProdi)
{
    try {
        if (empty($kodeProdi)) {
            return 100; // Nilai tinggi agar tidak mungkin lulus jika tidak ada prodi
        }

        $prodi = DB::table('program_studi')
                 ->where('kode_prodi', $kodeProdi)
                 ->select('passing_grade')
                 ->first();

        if ($prodi && $prodi->passing_grade !== null) {
            return (float)$prodi->passing_grade;
        }

        // Default passing grade berdasarkan mapping
        return $this->getDefaultPassingGrade($kodeProdi);

    } catch (\Exception $e) {
        Log::error('Error getting passing grade for prodi ' . $kodeProdi . ': ' . $e->getMessage());
        return $this->getDefaultPassingGrade($kodeProdi);
    }
}

    /**
     * Mapping nama jurusan ke kode prodi dari database
     * Ambil langsung dari tabel program_studi
     */
    private function getProdiMapping()
    {
        try {
            $prodiList = DB::table('program_studi')
                         ->where('is_aktif', 1)
                         ->select('kode_prodi', 'nama_prodi')
                         ->get();

            $mapping = [];
            foreach ($prodiList as $prodi) {
                $mapping[$prodi->nama_prodi] = $prodi->kode_prodi;
                
                // Tambahkan juga mapping untuk variasi nama yang mungkin
                $namaVariations = $this->getProdiNameVariations($prodi->nama_prodi);
                foreach ($namaVariations as $variation) {
                    $mapping[$variation] = $prodi->kode_prodi;
                }
            }

            // Tambahkan mapping khusus jika diperlukan
            $additionalMapping = [
                'Tek. Mesin' => '21201',
                'Tek. Elektro' => '26201',
                'Tek. Informatika' => '55201',
                'Sis. Informasi' => '57201',
                // Tambahkan mapping tambahan jika ada
            ];

            return array_merge($mapping, $additionalMapping);

        } catch (\Exception $e) {
            Log::error('Error getting prodi mapping from database: ' . $e->getMessage());
            
            // Fallback ke mapping statis jika database error
            return $this->getStaticProdiMapping();
        }
    }

    /**
     * Generate variasi nama prodi untuk mapping yang lebih baik
     */
    private function getProdiNameVariations($namaProdi)
    {
        $variations = [];
        
        // Hapus kata yang tidak perlu
        $cleanName = str_replace(['Program Studi', 'Prodi', 'S1', 'S2', 'S3'], '', $namaProdi);
        $cleanName = trim($cleanName);
        $variations[] = $cleanName;
        
        // Singkatan
        $words = explode(' ', $cleanName);
        if (count($words) > 1) {
            $abbreviation = '';
            foreach ($words as $word) {
                if (!empty($word)) {
                    $abbreviation .= substr($word, 0, 1);
                }
            }
            if (strlen($abbreviation) > 1) {
                $variations[] = $abbreviation;
            }
        }
        
        // Tanpa spasi
        $variations[] = str_replace(' ', '', $cleanName);
        
        return array_unique($variations);
    }

private function getKelulusanFromNilai($userId, $prodiPilihan1, $prodiPilihan2)
{
    try {
        // 1. Ambil data nilai quiz terbaru dari moodle
        $nilaiQuiz = $this->getNilaiQuizTerbaru($userId);
        
        if (!$nilaiQuiz || $nilaiQuiz['nilai'] === null) {
            return [
                'status' => 'Belum mengerjakan quiz',
                'nilai' => 0,
                'prodi_lulus' => '',
                'nama_prodi_lulus' => '',
                'pilihan_lulus' => '',
                'keterangan' => 'User belum mengerjakan quiz atau error mengambil data'
            ];
        }

        $nilai = $nilaiQuiz['nilai'];
        $quizId = $nilaiQuiz['quiz_id'];
        $attemptDate = $nilaiQuiz['attempt_date'] ?? null;

        // 2. Jika tidak ada prodi pilihan
        if (empty($prodiPilihan1) && empty($prodiPilihan2)) {
            return [
                'status' => 'Sudah mengerjakan quiz - Belum memilih prodi',
                'nilai' => $nilai,
                'prodi_lulus' => '',
                'nama_prodi_lulus' => '',
                'pilihan_lulus' => '',
                'keterangan' => 'Nilai: ' . $nilai . ' - Belum memilih program studi'
            ];
        }

        // 3. Ambil passing grade untuk masing-masing prodi
        $passingGradeP1 = $this->getPassingGrade($prodiPilihan1);
        $passingGradeP2 = $this->getPassingGrade($prodiPilihan2);

        // 4. Tentukan status kelulusan
        $lulusP1 = $nilai >= $passingGradeP1;
        $lulusP2 = $nilai >= $passingGradeP2;

        // 5. Ambil nama prodi untuk yang lulus
        $prodiLulus = '';
        $namaProdiLulus = '';
        $pilihanLulus = '';

        if ($lulusP1) {
            $prodiLulus = $prodiPilihan1;
            $pilihanLulus = 'Pilihan 1';
        } elseif ($lulusP2) {
            $prodiLulus = $prodiPilihan2;
            $pilihanLulus = 'Pilihan 2';
        }

        // Ambil nama prodi lulus
        if (!empty($prodiLulus)) {
            $prodiData = DB::table('program_studi')
                         ->where('kode_prodi', $prodiLulus)
                         ->select('nama_prodi')
                         ->first();
            $namaProdiLulus = $prodiData ? $prodiData->nama_prodi : $prodiLulus;
        }

        // 6. Generate hasil kelulusan
        if ($lulusP1) {
            return [
                'status' => 'LULUS',
                'nilai' => $nilai,
                'prodi_lulus' => $prodiLulus,
                'nama_prodi_lulus' => $namaProdiLulus,
                'pilihan_lulus' => $pilihanLulus,
                'passing_grade' => $passingGradeP1,
                'selisih' => $nilai - $passingGradeP1,
                'keterangan' => 'Lulus di ' . $namaProdiLulus . ' (Pilihan 1) - Nilai: ' . $nilai . ' | Passing: ' . $passingGradeP1,
                'attempt_date' => $attemptDate,
                'quiz_id' => $quizId
            ];
        } elseif ($lulusP2) {
            return [
                'status' => 'LULUS',
                'nilai' => $nilai,
                'prodi_lulus' => $prodiLulus,
                'nama_prodi_lulus' => $namaProdiLulus,
                'pilihan_lulus' => $pilihanLulus,
                'passing_grade' => $passingGradeP2,
                'selisih' => $nilai - $passingGradeP2,
                'keterangan' => 'Lulus di ' . $namaProdiLulus . ' (Pilihan 2) - Nilai: ' . $nilai . ' | Passing: ' . $passingGradeP2,
                'attempt_date' => $attemptDate,
                'quiz_id' => $quizId
            ];
        } else {
            // Tidak lulus di kedua pilihan
            $selisihP1 = $passingGradeP1 - $nilai;
            $selisihP2 = $passingGradeP2 - $nilai;
            
            return [
                'status' => 'TIDAK LULUS',
                'nilai' => $nilai,
                'prodi_lulus' => '',
                'nama_prodi_lulus' => '',
                'pilihan_lulus' => '',
                'passing_grade_p1' => $passingGradeP1,
                'passing_grade_p2' => $passingGradeP2,
                'selisih_p1' => $selisihP1,
                'selisih_p2' => $selisihP2,
                'keterangan' => 'Tidak lulus - Butuh ' . $selisihP1 . ' point untuk P1, ' . $selisihP2 . ' point untuk P2',
                'attempt_date' => $attemptDate,
                'quiz_id' => $quizId
            ];
        }

    } catch (\Exception $e) {
        Log::error('Error determining kelulusan from nilai for user_id ' . $userId . ': ' . $e->getMessage());
        return [
            'status' => 'Error menentukan kelulusan',
            'nilai' => 0,
            'prodi_lulus' => '',
            'nama_prodi_lulus' => '',
            'pilihan_lulus' => '',
            'keterangan' => 'Error: ' . $e->getMessage()
        ];
    }
}

                    'error' => $result['error'],

                    'course_id' => $courseId,
